---
title: 'User Info'
api: 'GET https://www.osintcat.net/api/user'
---

Retrieve information and usage statistics about your own account, including email OSINT credit information.

## Headers

<ParamField header="X-API-KEY" type="string" required>
  Your API key
</ParamField>

## Response

Returns account information including username, email, plan, usage statistics, and email OSINT credit information.

<ResponseField name="account_info" type="object">
  Account information object
  <ResponseField name="username" type="string">Your username</ResponseField>
  <ResponseField name="email" type="string">Your email address</ResponseField>
  <ResponseField name="plan" type="string">Your current plan (Free, Starter, Researcher, Investigator, or Enterprise)</ResponseField>
</ResponseField>

<ResponseField name="usage" type="object">
  Usage statistics object
  <ResponseField name="request_limit_daily" type="number">Daily request limit</ResponseField>
  <ResponseField name="requests_remaining_today" type="number">Remaining requests today</ResponseField>
  <ResponseField name="api_access" type="boolean">Whether API access is enabled</ResponseField>
  <ResponseField name="intelx_access" type="boolean">Whether IntelX access is enabled</ResponseField>
  <ResponseField name="max_ips" type="number">Maximum number of IP addresses allowed</ResponseField>
</ResponseField>

<ResponseField name="email_osint_credits" type="object">
  Email OSINT credit information
  <ResponseField name="current_balance" type="number">Current credit balance in EUR (2 decimal places)</ResponseField>
  <ResponseField name="price_per_search" type="number">Cost per search based on your plan in EUR (2 decimal places)</ResponseField>
  <ResponseField name="has_sufficient_credits" type="boolean">Whether you have enough credits for at least one search</ResponseField>
  <ResponseField name="searches_available" type="number">Estimated number of searches available with current balance (integer)</ResponseField>
</ResponseField>

```java Java
import java.net.HttpURLConnection;
import java.net.URL;
import java.io.BufferedReader;
import java.io.InputStreamReader;

public class OsintCatAPI {
    public static void main(String[] args) {
        try {
            String apiUrl = "https://www.osintcat.net/api/user";
            URL url = new URL(apiUrl);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            conn.setRequestProperty("X-API-KEY", "YOUR_API_KEY");
            
            int responseCode = conn.getResponseCode();
            if (responseCode == HttpURLConnection.HTTP_OK) {
                BufferedReader in = new BufferedReader(new InputStreamReader(conn.getInputStream()));
                String inputLine;
                StringBuilder response = new StringBuilder();
                while ((inputLine = in.readLine()) != null) {
                    response.append(inputLine);
                }
                in.close();
                System.out.println(response.toString());
            } else {
                System.out.println("Error: " + responseCode);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
