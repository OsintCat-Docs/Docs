---
title: 'User Info'
api: 'GET /api/user'
---

Retrieve information and usage statistics about your own account.

## Parameters

<ParamField body="query" type="string" required>
  Your API key
</ParamField>

## Response

Returns account information including username, email, plan, and usage statistics.

<ResponseField name="account_info" type="object">
  Account information object
  <ResponseField name="username" type="string">Your username</ResponseField>
  <ResponseField name="email" type="string">Your email address</ResponseField>
  <ResponseField name="plan" type="string">Your current plan (Free, Premium, or Enterprise)</ResponseField>
  <ResponseField name="member_since" type="string">Account creation date</ResponseField>
</ResponseField>

<ResponseField name="usage" type="object">
  Usage statistics object
  <ResponseField name="dashboard" type="object">
    Dashboard usage information
    <ResponseField name="request_limit_daily" type="string|number">Daily request limit for dashboard</ResponseField>
    <ResponseField name="requests_remaining_today" type="string|number">Remaining dashboard requests today</ResponseField>
  </ResponseField>
  <ResponseField name="api" type="object">
    API usage information
    <ResponseField name="request_limit_daily" type="string|number">Daily request limit for API</ResponseField>
    <ResponseField name="requests_made_today" type="string|number">API requests made today</ResponseField>
    <ResponseField name="requests_remaining_today" type="string|number">Remaining API requests today</ResponseField>
  </ResponseField>
  <ResponseField name="last_limit_reset" type="string">Timestamp of last limit reset (UTC)</ResponseField>
</ResponseField>

<ResponseField name="note" type="string">
  Information about when limits reset
</ResponseField>

```java Java
import java.net.HttpURLConnection;
import java.net.URL;
import java.io.BufferedReader;
import java.io.InputStreamReader;

public class OsintCatAPI {
    public static void main(String[] args) {
        try {
            String apiUrl = "https://www.osintcat.net/api/user?id=YOUR_API_KEY";
            URL url = new URL(apiUrl);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            
            int responseCode = conn.getResponseCode();
            if (responseCode == HttpURLConnection.HTTP_OK) {
                BufferedReader in = new BufferedReader(
                    new InputStreamReader(conn.getInputStream())
                );
                String inputLine;
                StringBuffer response = new StringBuffer();
                
                while ((inputLine = in.readLine()) != null) {
                    response.append(inputLine);
                }
                in.close();
                
                System.out.println(response.toString());
            } else {
                System.out.println("Error: " + responseCode);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
