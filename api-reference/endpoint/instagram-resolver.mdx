---
title: 'Instagram Resolver'
api: 'GET https://www.osintcat.net/api/instagram-resolver'
---

Resolve an Instagram share link to expose both the **author** of the post and the **sharer** (the user who generated the share link). Pulls the `xig_logged_out_dynamic_dialog_info` payload that Instagram embeds for logged-out viewers.

## Parameters

<ParamField query="link" type="string" required>
  An Instagram share URL, e.g. `https://www.instagram.com/share/p/ABCDEFGHIJ`
</ParamField>

## Headers

<ParamField header="X-API-KEY" type="string" required>
  Your API key
</ParamField>

## Response

```json
{
  "success": true,
  "author":  { "username": "...", "avatar": "..." },
  "sharer":  { "username": "...", "full_name": "...", "id": "...", "avatar": "..." },
  "entity_id": "...",
  "_meta": { ... }
}
```

```go Go
package main

import (
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "net/url"
)

func main() {
    baseURL := "https://www.osintcat.net/api/instagram-resolver"
    params := url.Values{}
    params.Add("link", "https://www.instagram.com/share/p/ABCDEFGHIJ")

    fullURL := baseURL + "?" + params.Encode()

    req, _ := http.NewRequest("GET", fullURL, nil)
    req.Header.Set("X-API-KEY", "YOUR_API_KEY")

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        fmt.Printf("Error: %v\n", err)
        return
    }
    defer resp.Body.Close()

    body, _ := io.ReadAll(resp.Body)
    var result map[string]interface{}
    json.Unmarshal(body, &result)
    fmt.Printf("%+v\n", result)
}
```

```java Java
import java.net.HttpURLConnection;
import java.net.URL;
import java.net.URLEncoder;
import java.io.BufferedReader;
import java.io.InputStreamReader;

public class OsintCatAPI {
    public static void main(String[] args) {
        try {
            String link = URLEncoder.encode("https://www.instagram.com/share/p/ABCDEFGHIJ", "UTF-8");
            String apiUrl = "https://www.osintcat.net/api/instagram-resolver?link=" + link;
            URL url = new URL(apiUrl);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            conn.setRequestProperty("X-API-KEY", "YOUR_API_KEY");

            int responseCode = conn.getResponseCode();
            if (responseCode == HttpURLConnection.HTTP_OK) {
                BufferedReader in = new BufferedReader(new InputStreamReader(conn.getInputStream()));
                String inputLine;
                StringBuilder response = new StringBuilder();
                while ((inputLine = in.readLine()) != null) {
                    response.append(inputLine);
                }
                in.close();
                System.out.println(response.toString());
            } else {
                System.out.println("Error: " + responseCode);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
