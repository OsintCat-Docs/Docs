---
title: 'Discord Monitor'
api: 'GET /api/discord-stalker'
---

Find messages, voice joins & leaves, name history, and deleted messages about a Discord user.

## Parameters

<ParamField body="query" type="string" required>
  A Discord user ID
</ParamField>

<ParamField body="id" type="string" required>
  Your API key
</ParamField>

## Response

Returns comprehensive Discord user monitoring data including message history, voice activity, name changes, and deleted messages.

```java Java
import java.net.HttpURLConnection;
import java.net.URL;
import java.io.BufferedReader;
import java.io.InputStreamReader;

public class OsintCatAPI {
    public static void main(String[] args) {
        try {
            String apiUrl = "https://www.osintcat.net/api/discord-stalker?query=123456789012345678&id=YOUR_API_KEY";
            URL url = new URL(apiUrl);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            
            int responseCode = conn.getResponseCode();
            if (responseCode == HttpURLConnection.HTTP_OK) {
                BufferedReader in = new BufferedReader(
                    new InputStreamReader(conn.getInputStream())
                );
                String inputLine;
                StringBuffer response = new StringBuffer();
                
                while ((inputLine = in.readLine()) != null) {
                    response.append(inputLine);
                }
                in.close();
                
                System.out.println(response.toString());
            } else {
                System.out.println("Error: " + responseCode);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
